  const off = -d.getTimezoneOffset(), s = off >= 0 ? '+' : '-', a = Math.abs(off);

/* ---------- colorimetría: sRGB → CIE L*a*b* (D65) → ITA° ---------- */
const LIN = new Float32Array(256);

function linToLab(r, g, b){
  const X = 0.4124564*r + 0.3575761*g + 0.1804375*b;
  const Y = 0.2126729*r + 0.7151522*g + 0.0721750*b;

function sampleCircle(ctx, W, H, cx, cy, r){
  const x0 = Math.max(0, Math.floor(cx - r)), x1 = Math.min(W, Math.ceil(cx + r));
  const y0 = Math.max(0, Math.floor(cy - r)), y1 = Math.min(H, Math.ceil(cy + r));
  const w = x1 - x0, h = y1 - y0;
  const out = { n: 0, sat: 0, dark: 0 };
  if (w <= 0 || h <= 0) return out;
  const d = ctx.getImageData(x0, y0, w, h).data;
  const cap = w*h;
  const R = new Float32Array(cap), G = new Float32Array(cap), B = new Float32Array(cap), Yv = new Float32Array(cap);
  const r2 = r*r;
  let n = 0, sat = 0, dark = 0;
  for (let y = 0; y < h; y++){
    const dy = y0 + y + 0.5 - cy;
    for (let x = 0; x < w; x++){
      const dx = x0 + x + 0.5 - cx;
      if (dx*dx + dy*dy > r2) continue;
      const i = (y*w + x) * 4;
      const r8 = d[i], g8 = d[i+1], b8 = d[i+2];
      const mx = Math.max(r8, g8, b8);
      if (mx >= 250) sat++;
      if (mx <= 10) dark++;
      const lr = LIN[r8], lg = LIN[g8], lb = LIN[b8];
      R[n] = lr; G[n] = lg; B[n] = lb;
      Yv[n] = 0.2126729*lr + 0.7151522*lg + 0.0721750*lb;
      n++;
    }
  }
  Object.assign(out, { n, sat, dark, R, G, B, Y: Yv });
  return out;
}

// Media robusta en espacio lineal: descarta el 10 % más oscuro y el 10 % más brillante
// (vello, poros, brillos especulares).
function robustMean(s){
  const n = s.n;
  if (!n) return null;
  const sorted = s.Y.slice(0, n).sort();
  const lo = sorted[Math.floor(n*0.10)], hi = sorted[Math.min(n-1, Math.floor(n*0.90))];
  let r = 0, g = 0, b = 0, c = 0, sL = 0, sL2 = 0;
  for (let i = 0; i < n; i++){
    const y = s.Y[i];
    if (y < lo || y > hi) continue;
    r += s.R[i]; g += s.G[i]; b += s.B[i];
    const L = LfromY(y); sL += L; sL2 += L*L; c++;
  }
  if (!c) return null;
  const mL = sL/c;
  return { r: r/c, g: g/c, b: b/c, kept: c, sdL: Math.sqrt(Math.max(0, sL2/c - mL*mL)) };
}

    } else {
      if (gs.sat/gs.n > 0.005) flags.push('tarjeta_saturada');
      if (gray.sdL > 4) flags.push('tarjeta_no_uniforme');
      // Corrección diagonal (von Kries) en RGB lineal: la tarjeta 18 % pasa a ser gris neutro de reflectancia 0,18
      gains = [0.18/gray.r, 0.18/gray.g, 0.18/gray.b];
      if (gains.some(g => g > 5 || g < 0.2)) flags.push('correccion_extrema');
      grayUsed = true;
    }
  } else {
    flags.push('sin_tarjeta_gris');
  }
  const raw = linToLab(sk.r, sk.g, sk.b);
  const lab = linToLab(sk.r*gains[0], sk.g*gains[1], sk.b*gains[2]);
  if (lab.b <= 1) flags.push('b_bajo_ita_inestable');
  return {
    ok: true, L: lab.L, a: lab.a, b: lab.b, ITA: itaDeg(lab.L, lab.b),
    ITA_uncorrected: itaDeg(raw.L, raw.b), grayUsed, gains, flags,
    nPixels: skin.n, sdL: sk.sdL, W, H
  };
}

function clampPos(pos, r, W, H){
  const m = Math.min(W, H), rx = r*m/W, ry = r*m/H;
  return { x: Math.min(1-rx, Math.max(rx, pos.x)), y: Math.min(1-ry, Math.max(ry, pos.y)) };
}

async function startCamera(){
  if (cam.stream) return;
  if (!window.isSecureContext || !navigator.mediaDevices || !navigator.mediaDevices.getUserMedia){
    stageMsg('La cámara en vivo necesita HTTPS. Usa “Usar app de cámara” (con el flash apagado).');
    $('#btn-capture').disabled = true; $('#btn-lock').disabled = true;
    renderCaps(); return;
  }
  stageMsg('Abriendo cámara…');
  try {
    cam.stream = await navigator.mediaDevices.getUserMedia({ audio: false, video: {
      facingMode: { ideal: 'environment' }, width: { ideal: 1920 }, height: { ideal: 1080 } } });
  } catch (err){
    stageMsg(err && err.name === 'NotAllowedError'
      ? 'Permiso de cámara denegado. Actívalo en el navegador o usa “Usar app de cámara”.'
      : 'No se pudo abrir la cámara (' + (err && err.name || 'error') + '). Usa “Usar app de cámara”.');
    $('#btn-capture').disabled = true; $('#btn-lock').disabled = true;
    renderCaps(); return;
  }
  if (!$('#v-cam').classList.contains('on')){ stopCamera(); return; }
  cam.track = cam.stream.getVideoTracks()[0];
  cam.caps = cam.track.getCapabilities ? cam.track.getCapabilities() : {};
  video.srcObject = cam.stream;
  try { await video.play(); } catch(e){}
  await new Promise(res => video.videoWidth ? res() : video.addEventListener('loadedmetadata', res, { once: true }));
  stageMsg('');
  $('#btn-capture').disabled = false;
  if (cam.caps.torch){ try { await cam.track.applyConstraints({ advanced: [{ torch: false }] }); } catch(e){} }
  cam.locked = false;
  if (cfg.reapplyLock && cfg.camLock){ cam.locked = await applyLock(cfg.camLock); }
  readSettings(); renderCaps(); drawOverlay();
}
function stopCamera(){
  if (cam.stream){ cam.stream.getTracks().forEach(t => t.stop()); }
  cam.stream = null; cam.track = null; cam.locked = false;
  video.srcObject = null;
}

}
async function applyLock(vals){
  const c = cam.caps, adv = {};
  if (hasMode(c.exposureMode, 'manual')){
    adv.exposureMode = 'manual';
    if (vals.exposureTime != null && c.exposureTime) adv.exposureTime = clampCap(vals.exposureTime, c.exposureTime);
    if (vals.iso != null && c.iso) adv.iso = clampCap(vals.iso, c.iso);
  }
  if (hasMode(c.whiteBalanceMode, 'manual')){
    adv.whiteBalanceMode = 'manual';
    if (vals.colorTemperature != null && c.colorTemperature) adv.colorTemperature = clampCap(vals.colorTemperature, c.colorTemperature);
  }
  if (!Object.keys(adv).length) return false;
  try { await cam.track.applyConstraints({ advanced: [adv] }); return true; }
  catch(e){
    try { await cam.track.applyConstraints({ advanced: [{ exposureMode: adv.exposureMode, whiteBalanceMode: adv.whiteBalanceMode }] }); return true; }
    catch(e2){ return false; }
  }
}

const mean = a => a.reduce((s, v) => s + v, 0) / a.length;
const sd = a => a.length < 2 ? null : Math.sqrt(a.reduce((s, v) => s + (v - mean(a))**2, 0) / (a.length - 1));
function buildRecord(){
  const S = cur.shots, fz = fitzResult();
  const itas = S.map(s => s.ITA);
  const ITA_mean = mean(itas), ITA_sd = sd(itas);
  const flags = new Set(); S.forEach(s => s.flags.forEach(f => flags.add(f)));
  if (ITA_sd != null && ITA_sd > 3) flags.add('variabilidad_alta');
  const cat = itaCat(ITA_mean);
  const c0 = (S[0] && S[0].cam) || {};
  const rec = {
    uid: cur.uid,
    participant_id: cur.participant_id,
    timestamp: cur.timestamp,
    photo_time_first: S[0] ? S[0].time : '',
    operator: cfg.operator,
    device_label: cfg.device,
    body_site: cfg.site,
    n_photos: S.length,
    L_mean: mean(S.map(s => s.L)), a_mean: mean(S.map(s => s.a)), b_mean: mean(S.map(s => s.b)),
    ITA_mean, ITA_sd,
    ITA_per_photo: itas.map(v => v.toFixed(2)).join(';'),
    ITA_category: cat.code, ITA_category_es: cat.es,
    ITA_uncorrected_mean: mean(S.map(s => s.ITA_uncorrected)),
    gray_card_used: S.every(s => s.grayUsed),
    wb_gains_rgb: S.map(s => s.gains.map(g => g.toFixed(3)).join('/')).join(';'),
    capture_source: [...new Set(S.map(s => s.source))].join(';'),
    cam_locked: S.every(s => s.cam && s.cam.locked),
    cam_exposure_mode: c0.exposureMode, cam_exposure_time: c0.exposureTime, cam_iso: c0.iso,
    cam_wb_mode: c0.whiteBalanceMode, cam_color_temp: c0.colorTemperature,
    image_size: S[0] ? `${S[0].W}x${S[0].H}` : '',
    quality_flags: [...flags].join(';'),
    fitz_score: fz.score, fitz_type: fz.type ? ROMAN[fz.type-1] : '', fitz_type_num: fz.type,
    notes: $('#sum-notes').value.trim(),
    user_agent: navigator.userAgent,
    app_version: APP_VERSION
  };
  cur.fitz.forEach((v, i) => rec['fitz_q' + (i+1)] = v);
  return rec;
}

const CSV_COLS = ['participant_id','timestamp','photo_time_first','operator','device_label','body_site','n_photos',
  'L_mean','a_mean','b_mean','ITA_mean','ITA_sd','ITA_per_photo','ITA_category','ITA_category_es','ITA_uncorrected_mean',

function csvCell(v){
  if (v == null) return '';
  if (typeof v === 'number') return isFinite(v) ? (Number.isInteger(v) ? String(v) : v.toFixed(3)) : '';
  if (typeof v === 'boolean') return v ? 'true' : 'false';
  const s = String(v);
  return /[",\n\r]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
}
function exportCSV(){
  if (!records.length){ alert('No hay registros para exportar.'); return; }
  const lines = [CSV_COLS.join(',')].concat(records.map(r => CSV_COLS.map(c => csvCell(r[c])).join(',')));
  const blob = new Blob(['﻿' + lines.join('\r\n') + '\r\n'], { type: 'text/csv;charset=utf-8' });
  const d = new Date(), p = n => String(n).padStart(2, '0');
  const name = `ita_fitzpatrick_${cfg.device ? cfg.device.replace(/[^\w-]+/g, '_') + '_' : ''}${d.getFullYear()}${p(d.getMonth()+1)}${p(d.getDate())}_${p(d.getHours())}${p(d.getMinutes())}.csv`;
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob); a.download = name;
  document.body.appendChild(a); a.click(); a.remove();
  setTimeout(() => URL.revokeObjectURL(a.href), 4000);
  records.forEach(r => r._exported = true); saveRecords(); renderHome();
}
